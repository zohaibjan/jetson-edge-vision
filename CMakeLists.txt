cmake_minimum_required(VERSION 3.20)
project(jetson_edge_vision VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(BUILD_DEEPSTREAM "Build proprietary-SDK adapters on the target device" OFF)
add_library(edge_decode STATIC cpp/src/decode.cpp)
target_include_directories(edge_decode PUBLIC cpp/include)
target_compile_options(edge_decode PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_executable(test_decode cpp/tests/test_decode.cpp)
target_link_libraries(test_decode PRIVATE edge_decode)
add_test(NAME yolo_decode COMMAND test_decode)
if(BUILD_DEEPSTREAM)
  set(DEEPSTREAM_ROOT "/opt/nvidia/deepstream/deepstream" CACHE PATH "Installed SDK")
  find_package(PkgConfig REQUIRED)
  find_package(Threads REQUIRED)
  pkg_check_modules(GST REQUIRED IMPORTED_TARGET gstreamer-1.0 gstreamer-base-1.0)
  pkg_check_modules(ZMQ REQUIRED IMPORTED_TARGET libzmq)
  pkg_check_modules(JSON REQUIRED IMPORTED_TARGET json-glib-1.0)
  find_path(NVDS_INCLUDE nvdsmeta.h HINTS "${DEEPSTREAM_ROOT}/sources/includes" REQUIRED)
  find_path(TRT_INCLUDE NvInfer.h HINTS /usr/include/aarch64-linux-gnu /usr/include/x86_64-linux-gnu REQUIRED)
  find_library(NVDS_META nvds_meta HINTS "${DEEPSTREAM_ROOT}/lib" REQUIRED)
  find_library(NVDS_GST_META nvdsgst_meta HINTS "${DEEPSTREAM_ROOT}/lib" REQUIRED)
  add_library(nvdsinfer_edge_yolo SHARED cpp/plugins/yolo_parser.cpp)
  target_include_directories(nvdsinfer_edge_yolo PRIVATE ${NVDS_INCLUDE} ${TRT_INCLUDE})
  target_link_libraries(nvdsinfer_edge_yolo PRIVATE edge_decode)
  add_library(gstedgevision SHARED cpp/plugins/edge_yolo.cpp cpp/plugins/edge_events.c cpp/src/publisher.cpp)
  target_include_directories(gstedgevision PRIVATE cpp/include ${NVDS_INCLUDE} ${TRT_INCLUDE})
  target_link_libraries(gstedgevision PRIVATE PkgConfig::GST PkgConfig::ZMQ PkgConfig::JSON ${NVDS_META} ${NVDS_GST_META} Threads::Threads m)
  target_compile_definitions(gstedgevision PRIVATE PACKAGE="jetson-edge-vision" DS_NEW_BUFAPI=1)
  set_target_properties(gstedgevision PROPERTIES BUILD_RPATH "${DEEPSTREAM_ROOT}/lib")
endif()
